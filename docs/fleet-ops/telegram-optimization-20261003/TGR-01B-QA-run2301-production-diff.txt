+def _routed_profile_for_override(runner: Any, source: Any) -> Optional[str]:
+    """The profile an inbound source is routed to, for channel-override scoping only.
+
+    The pinned :class:`~gateway.session_identity.RoutingIdentity` is authoritative; a bare
+    ``source.profile`` (a routing result without a pinned identity) is the fallback.
+    ``None`` means the launcher/primary scope."""
+    if source is None:
+        return None
+    from gateway.session_identity import identity_of
+
+    identity = identity_of(source)
+    if identity is not None and identity.multiplexed:
+        return identity.runtime_profile or None
+    profile = getattr(source, "profile", None)
+    return profile.strip() if isinstance(profile, str) and profile.strip() else None
+

+def _get_channel_override_for_source(runner: Any, config: Any, source: Any) -> Optional[Any]:
+    """``_get_channel_override`` scoped to the config that actually owns *source*.
+
+    ``GatewayRunner.config`` is the LAUNCHER profile's typed config; under multiplex a routed
+    (secondary) profile's chats must match ``channel_overrides`` in THAT profile's own served
+    config from ``_profile_configs`` (populated/refreshed in run_adapters), never the
+    launcher's rules. A missing/unknown/unserved routed profile falls back to NO override
+    (the global default) — it must not borrow another profile's rule. Primary/standalone
+    sources keep the launcher config. ``system_prompt`` resolution is deliberately untouched
+    (different contract, keyed by platform/chat without a routed source)."""
+    from gateway.run import _get_channel_override
+
+    platform = getattr(source, "platform", None)
+    if source is None or platform is None:
+        return None
+    chat_id = str(getattr(source, "chat_id", "") or "")
+
+    def _lookup(cfg):
+        return _get_channel_override(
+            cfg, platform, chat_id,
+            thread_id=str(source.thread_id) if getattr(source, "thread_id", None) else None,
+            parent_id=str(source.parent_chat_id) if getattr(source, "parent_chat_id", None) else None,
+        )
+
+    routed = _routed_profile_for_override(runner, source)
+    primary = getattr(runner, "_primary_profile_name", None) or "default"
+    if not routed or routed == primary:
+        return _lookup(config)
+    return _lookup((getattr(runner, "_profile_configs", None) or {}).get(routed))
